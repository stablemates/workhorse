## Transactional side effect

Create a record and its follow-up work atomically.

```ts
import { Pool } from "pg";
import { Queue } from "@workhorse/core";

const pool = new Pool({ connectionString: process.env.DATABASE_URL });
const queue = new Queue(pool);

export async function createOrder(input: { customerId: string; total: number }) {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");
    const { rows } = await client.query<{ id: string }>(
      "INSERT INTO orders (customer_id, total) VALUES ($1, $2) RETURNING id",
      [input.customerId, input.total],
    );
    const orderId = rows[0]!.id;

    await queue.enqueue(
      "order.fulfill",
      { orderId },
      { queue: "orders", idempotency: { key: `fulfill:${orderId}` } },
      client,
    );

    await client.query("COMMIT");
    return orderId;
  } catch (error) {
    await client.query("ROLLBACK");
    throw error;
  } finally {
    client.release();
  }
}
```

## Durable multi-stage pipeline

Checkpoints replay completed stages; the wait releases the lease between them.

```ts
worker.handle("order.fulfill", async (payload: { orderId: string }, context) => {
  const charge = await context.checkpoint("charge", () =>
    payments.charge(payload.orderId, { idempotencyKey: `charge:${payload.orderId}` }),
  );

  const shipment = await context.checkpoint("ship", () =>
    logistics.createShipment(payload.orderId, charge.id),
  );

  // Release the worker slot while the carrier scans the parcel.
  await context.sleep("carrier-scan-window", 6 * 60 * 60_000);

  const tracking = await context.checkpoint("track", () => logistics.track(shipment.id));

  return { chargeId: charge.id, shipmentId: shipment.id, tracking };
});
```

## Cancellation-aware long job

```ts
worker.handle("export.build", async (payload: { exportId: string }, context) => {
  const rows = await context.checkpoint("query", () => warehouse.select(payload.exportId));

  for (const [index, batch] of chunk(rows, 500).entries()) {
    if (context.signal.aborted) throw context.signal.reason;
    await context.checkpoint(`upload-${index}`, () => storage.putBatch(payload.exportId, batch));
  }

  return { exportId: payload.exportId, batches: Math.ceil(rows.length / 500) };
});
```

Request cancellation from your operator layer after checking permissions:

```ts
await queue.cancel(jobId, { requestedBy: actor.email, reason: "export superseded" });
```

## Fan-out with a bounded batch

```ts
const recipients = await loadRecipients(campaignId); // up to 1,000 per call

for (const page of chunk(recipients, 1_000)) {
  await queue.enqueueMany(
    page.map((recipient) => ({
      type: "campaign.send",
      payload: { campaignId, recipientId: recipient.id },
      options: {
        queue: "bulk",
        tags: [`campaign:${campaignId}`],
        idempotency: { key: `campaign:${campaignId}:${recipient.id}`, ttlMs: 7 * 86_400_000 },
        retryPolicy: {
          type: "exponential",
          initialDelayMs: 5_000,
          multiplier: 2,
          maxDelayMs: 600_000,
        },
      },
    })),
  );
}
```

The idempotency keys make the whole fan-out safe to re-run after a partial failure.

## Incident redrive runbook

```ts
const filter = { queue: "billing", errorName: "ProviderTimeout", finishedAfter: incidentStart };
const request = {
  requestedBy: "oncall@example.com",
  reason: "provider incident 2026-08-03 resolved",
  requestId: "incident-2026-08-03:billing",
};

// 1. Preview.
const preview = await queue.redriveMany(filter, request, { dryRun: true, limit: 100 });
console.log(`${preview.results.length} eligible`);

// 2. Drain the backlog page by page.
let cursor;
do {
  const page = await queue.redriveMany(filter, request, { limit: 500, cursor });
  cursor = page.nextCursor ?? undefined;
} while (cursor);
```

## Testing handlers

`runOnce()` executes a single claim pass, which makes integration tests deterministic.

```ts
import { installSchema, Queue, Worker } from "@workhorse/core";

const queue = new Queue(pool);
await installSchema(pool);

const jobId = await queue.enqueue("greet", { name: "Ada" });

const worker = new Worker(queue).handle("greet", async (payload: { name: string }) => ({
  greeting: `Hello, ${payload.name}`,
}));

await worker.runOnce();

const snapshot = await queue.getJob(jobId);
expect(snapshot?.state).toBe("succeeded");
```

For crash-boundary tests, `WorkerOptions.failpoint` deterministically throws at `afterClaim`,
`beforeHandler`, `afterHandler`, `beforeComplete`, or `afterComplete`:

```ts
const crashing = new Worker(queue, { failpoint: "beforeComplete" });
```

Injected crashes deliberately bypass normal fail/retry handling so you can exercise lease recovery.
