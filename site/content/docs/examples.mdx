## Enqueue a task in the transaction of an API route

An API request creates an order. The order needs a fulfillment task, but only if the order commits.
To get this result, enqueue the task in the transaction that inserts the order. Then the task and
the order commit together, or they roll back together.

- In TypeScript, give the transaction client as the last argument to `enqueue`.
- In Python, Go, Rust, and Ruby, make the `Queue` on the open transaction, or on the connection
  that holds it.

<Tabs items={["TypeScript", "Python", "Go", "Rust", "Ruby"]}>
  <Tab value="TypeScript">
    ```ts
    import { Admin, Pool, Queue } from "@stablemates/workhorse";

    const pool = new Pool({ connectionString: process.env.DATABASE_URL });
    const queue = new Queue(pool);
    const admin = new Admin(pool);

    export async function createOrder(orderId: string, items: string[]) {
      const client = await pool.connect();
      try {
        await client.query("BEGIN");
        await client.query("INSERT INTO orders (id, items) VALUES ($1, $2)", [orderId, items]);
        await queue.enqueue("order.fulfill", { orderId }, { queue: "orders" }, client);
        await client.query("COMMIT");
      } catch (error) {
        await client.query("ROLLBACK");
        throw error;
      } finally {
        client.release();
      }
    }
    ```

  </Tab>

    ```go verify
    package transaction

    import (
    	"context"

    	"github.com/jackc/pgx/v5/pgxpool"
    	workhorse "github.com/stablemates/workhorse/go"
    )

    func createOrder(ctx context.Context, pool *pgxpool.Pool, orderID string, items []string) error {
    	tx, err := pool.Begin(ctx)
    	if err != nil {
    		return err
    	}
    	defer tx.Rollback(ctx)

    	if _, err := tx.Exec(ctx,
    		"INSERT INTO orders (id, items) VALUES ($1, $2)",
    		orderID,
    		items,
    	); err != nil {
    		return err
    	}
    	queue := workhorse.NewQueue(workhorse.NewPGXExecutor(tx), "orders")
    	if _, err := queue.Enqueue(ctx, "order.fulfill", map[string]any{
    		"orderId": orderID,
    	}, workhorse.EnqueueOptions{}); err != nil {
    		return err
    	}
    	return tx.Commit(ctx)
    }
    ```

    ```rust
        order_id: &str,
        items: &[String],
    ) -> Result<(), Box<dyn std::error::Error>> {

        let transaction = client.transaction().await?;

            .await?;
        Queue::new(&transaction, "orders")

            .await?;
        transaction.commit().await?;
        Ok(())
    }
    ```

If the insert fails, no task exists. If the enqueue fails, no order exists. At no time can a reader
see one without the other. A TypeScript application that uses Drizzle, Prisma, TypeORM, or Kysely
gets the same guarantee. It uses the `forTransaction` boundary of the adapter package.

## Send two emails a week apart with checkpoints and a durable sleep

A trial signup sends a welcome email now and a follow-up email in seven days. The reader must not
get an email two times. A checkpoint is a named piece of handler code. Workhorse stores its result,
so a later run of the handler does not run the code again.

- Put each send in a checkpoint.
- Wait with a durable sleep. The task uses no worker slot during the week.
- Give the mail provider an idempotency key that you make from the task ID. A checkpoint alone
  cannot stop a repeated send.

<Tabs items={["TypeScript", "Python", "Go", "Rust", "Ruby"]}>
  <Tab value="TypeScript">
    ```ts
    worker.handle("trial.lifecycle", async (payload: { to: string; followUpAt: string }, context) => {
      await context.checkpoint("welcome", () =>
        mailer.welcome(payload.to, { idempotencyKey: `welcome:${context.task.id}` }),
      );

      await context.sleepUntil("follow-up-window", new Date(payload.followUpAt));

      await context.checkpoint("follow-up", () =>
        mailer.followUp(payload.to, { idempotencyKey: `follow-up:${context.task.id}` }),
      );
      return { deliveredTo: payload.to };
    });
    ```

  </Tab>
  <Tab value="Python">
    ```python verify
    from datetime import datetime
    from typing import TypedDict

    from workhorse import HandlerContext, Json, Worker


    class Trial(TypedDict):
        to: str
        followUpAt: str


    # Stand-ins for a mail provider. A real send passes the key to the provider's
    # idempotency option, and the provider drops a repeated send with the same key.
    def send_welcome(to: str, *, idempotency_key: str) -> dict[str, Json]:
        return {"deliveredTo": to, "kind": "welcome", "idempotencyKey": idempotency_key}


    def send_follow_up(to: str, *, idempotency_key: str) -> dict[str, Json]:
        return {"deliveredTo": to, "kind": "follow-up", "idempotencyKey": idempotency_key}


    def register_trial_handler(worker: Worker) -> None:
        def trial_lifecycle(payload: Trial, context: HandlerContext) -> dict[str, Json]:
            to = payload["to"]
            follow_up_at = payload["followUpAt"]
            context.checkpoint(
                "welcome",
                lambda: send_welcome(to, idempotency_key=f"welcome:{context.task.id}"),
            )
            context.sleep_until("follow-up-window", datetime.fromisoformat(follow_up_at))
            context.checkpoint(
                "follow-up",
                lambda: send_follow_up(to, idempotency_key=f"follow-up:{context.task.id}"),
            )
            return {"deliveredTo": to}

        worker.handle("trial.lifecycle", trial_lifecycle)
    ```

    ```go verify
    package trial

    import (
    	"context"
    	"fmt"
    	"time"

    	workhorse "github.com/stablemates/workhorse/go"
    )

    // Stand-ins for a mail provider. A real send passes the key to the provider's
    // idempotency option, and the provider drops a repeated send with the same key.
    func sendWelcome(to, idempotencyKey string) (any, error) {
    	return map[string]any{"deliveredTo": to, "kind": "welcome", "idempotencyKey": idempotencyKey}, nil
    }

    func sendFollowUp(to, idempotencyKey string) (any, error) {
    	return map[string]any{"deliveredTo": to, "kind": "follow-up", "idempotencyKey": idempotencyKey}, nil
    }

    func registerTrialHandler(worker *workhorse.Worker) {
    	worker.Handle("trial.lifecycle", func(
    		_ context.Context,
    		payload any,
    		handler *workhorse.HandlerContext,
    	) (any, error) {
    		message, ok := payload.(map[string]any)
    		if !ok {
    			return nil, fmt.Errorf("trial payload must be an object")
    		}
    		to, ok := message["to"].(string)
    		if !ok {
    			return nil, fmt.Errorf("trial payload needs a string to field")
    		}
    		followUpText, ok := message["followUpAt"].(string)
    		if !ok {
    			return nil, fmt.Errorf("trial payload needs a string followUpAt field")
    		}
    		followUpAt, err := time.Parse(time.RFC3339, followUpText)
    		if err != nil {
    			return nil, err
    		}

    		if _, err := handler.Checkpoint("welcome", func() (any, error) {
    			return sendWelcome(to, "welcome:"+handler.Task.ID)
    		}); err != nil {
    			return nil, err
    		}
    		if err := handler.SleepUntil("follow-up-window", followUpAt); err != nil {
    			return nil, err
    		}
    		if _, err := handler.Checkpoint("follow-up", func() (any, error) {
    			return sendFollowUp(to, "follow-up:"+handler.Task.ID)
    		}); err != nil {
    			return nil, err
    		}
    		return map[string]any{"deliveredTo": to}, nil
    	})
    }
    ```

    ```rust
    use serde::Deserialize;
    use serde_json::{json, Value};
    use workhorse::{HandlerError, Worker};

    #[derive(Deserialize)]
    #[serde(rename_all = "camelCase")]
    struct Trial {
        to: String,
        follow_up_at: DateTime<Utc>,
    }

    // Stand-ins for a mail provider. A real send passes the key to the provider's
    // idempotency option, and the provider drops a repeated send with the same key.
    async fn send_welcome(to: &str, idempotency_key: &str) -> Result<Value, HandlerError> {
        Ok(json!({ "deliveredTo": to, "kind": "welcome", "idempotencyKey": idempotency_key }))
    }

    async fn send_follow_up(to: &str, idempotency_key: &str) -> Result<Value, HandlerError> {
        Ok(json!({ "deliveredTo": to, "kind": "follow-up", "idempotencyKey": idempotency_key }))
    }

    pub fn register_trial_handler(worker: &Worker) {
        worker.handle("trial.lifecycle", |trial: Trial, context| async move {
            let welcome_key = format!("welcome:{}", context.task().id);
            let _: Value =
                context.checkpoint("welcome", || send_welcome(&trial.to, &welcome_key)).await?;
            context.sleep_until("follow-up-window", trial.follow_up_at).await?;
            let follow_up_key = format!("follow-up:{}", context.task().id);
            let _: Value = context
                .checkpoint("follow-up", || send_follow_up(&trial.to, &follow_up_key))
                .await?;
            Ok(json!({ "deliveredTo": trial.to }))
        });
    }
    ```

The sleep stores a named timer, and the worker releases the task. The task waits in the state
`scheduled`, and it uses no worker slot for seven days. When the timer ends, a worker runs the
handler again from the start. The `welcome` checkpoint returns its stored value and does not send
again.

A checkpoint saves its value only after the send returns. Thus, a crash at the wrong time can
repeat a send.

**Example.** The process stops between the send and the save.

1. The handler sends the welcome email, and the provider accepts it.
2. The process stops before the `welcome` checkpoint saves its value.
3. The next attempt finds no saved value, so it sends the email again.
4. The key `welcome:<task ID>` is the same on each attempt, so the provider drops the second send.

Without the key, the reader can get the email two times.
[Protect an external call that a checkpoint cannot protect](/docs/concepts#protect-an-external-call-that-a-checkpoint-cannot-protect)
explains this case.

## Run a nightly task from a schedule that you sync at deployment

An invoice run must occur each night at 03:00. You do not want a separate scheduler service. Write
the schedule in your code, and sync it during the deployment. Each worker that watches the
namespace of the schedule can start the task.

<Tabs items={["TypeScript", "Python", "Go", "Rust", "Ruby"]}>
  <Tab value="TypeScript">
    ```ts
    // Deployment step: reconcile the namespace to exactly this set.
    await queue.syncSchedules("billing-production", [
      {
        name: "nightly-invoice-run",
        schedule: "0 3 * * *",
        task: {
          type: "invoice.generate",
          queue: "billing",
          payload: { scope: "due" },
        },
      },
    ]);

    // Worker process: evaluate and fire schedules for that namespace.
    const worker = new Worker(queue, {
      scheduleNamespaces: ["billing-production"],
    });
    worker.handle("invoice.generate", generateInvoices);
    await worker.run();
    ```

  </Tab>
  <Tab value="Python">
    ```python verify
    from typing import TypedDict

    import psycopg
    from psycopg_pool import ConnectionPool

    from workhorse import (
        HandlerContext,
        Json,
        Queue,
        ScheduleDefinition,
        ScheduledTask,
        Worker,
    )


    class InvoiceRun(TypedDict):
        scope: str


    def generate_invoices(payload: InvoiceRun, _context: HandlerContext) -> dict[str, Json]:
        return {"generated": True, "scope": payload["scope"]}


    def run_billing_worker(database_url: str) -> None:
        # Deployment step: reconcile the namespace to exactly this set.
        with psycopg.connect(database_url) as connection:
            Queue(connection).sync_schedules(
                "billing-production",
                (
                    ScheduleDefinition(
                        name="nightly-invoice-run",
                        schedule="0 3 * * *",
                        task=ScheduledTask(
                            type="invoice.generate",
                            queue="billing",
                            payload={"scope": "due"},
                        ),
                    ),
                ),
            )

        # Worker process: evaluate and fire schedules for that namespace.
        with ConnectionPool(
            database_url, min_size=3, max_size=3, kwargs={"autocommit": True}
        ) as pool:
            worker = Worker(
                pool,
                queue="billing",
                schedule_namespaces=("billing-production",),
            ).handle("invoice.generate", generate_invoices)
            worker.run()
    ```

  </Tab>

    ```rust
        Ok(())
    ```

Two workers cannot start the same occurrence two times. Each occurrence has a durable key: the
namespace, the name, and the planned time. If many workers try at the same time, PostgreSQL makes
one task. If you remove the definition from the array, the next deployment disables the schedule.
Thus, your code controls which schedules exist.

## Make one task for each webhook event

A payment provider can send the same webhook event more than one time. Each event must make exactly
one task. Make the idempotency key from the event ID of the provider. A repeated delivery then
returns the first task and does not make a new task.

<Tabs items={["TypeScript", "Python", "Go", "Rust", "Ruby"]}>
  <Tab value="TypeScript">
    ```ts
    export async function handleStripeWebhook(event: { id: string; type: string; data: unknown }) {
      const taskId = await queue.enqueue(
        "stripe.event",
        { eventId: event.id, eventType: event.type },
        {
          queue: "webhooks",
          idempotency: {
            key: `stripe:${event.id}`,
            scope: "stripe-webhooks",
          },
        },
      );
      return { accepted: taskId };
    }
    ```

  </Tab>

An equal repeated request returns the same task ID. It makes no new task, event, or notification.
Thus, the provider can safely retry the route.

If a request uses the key again with a _different_ payload, each SDK gives an error:

- TypeScript and Python throw `EnqueueIdempotencyConflictError`.
- Go returns the matching typed error.
- Rust returns `Error::EnqueueIdempotencyConflict`.
- Ruby raises `EnqueueIdempotencyConflictError`.

No SDK ignores the difference. The handler still runs at least one time, and sometimes more. Thus,
its external effects must also be safe to repeat.

## Redrive the failed tasks of an incident

A provider outage moved many tasks to the dead letters. A dead letter is a task that failed
permanently. The provider works again, and an operator must redrive each of those tasks one time.
A redrive makes a new task from a failed task. Like each task, the new task runs at least one time.

1. Do a dry run first.
2. Use one request ID for the incident on all pages. Then, if the script crashes, you can run it
   again without duplicate work.

<Tabs items={["TypeScript", "Python", "Go", "Rust", "Ruby"]}>
  <Tab value="TypeScript">
    ```ts
    const filter = { queue: "billing", errorName: "ProviderTimeout" };
    const request = {
      actor: "reviewer@example.com",
      reason: "provider incident INC-2041 resolved",
      requestId: "INC-2041",
    };

    // 1. Preview what would be redriven. Creates nothing.
    const preview = await admin.redriveMany(filter, request, { dryRun: true, limit: 100 });
    console.log(`${preview.results.length} tasks eligible`);

    // 2. Redrive for real, page by page, reusing the same request.
    let cursor = undefined;
    do {
      const page = await admin.redriveMany(filter, request, { limit: 100, cursor });
      for (const result of page.results) {
    ```

Each redrive creates a new task. The new task copies the queue, the type, the payload, and the
retry policy. The failed task stays unchanged as evidence. Each result names the failed task and
the new task. If you repeat a page with the same request ID, Workhorse returns the existing new
tasks. Thus, you can safely run the script again after a crash.

## Cancel a task that is running

A customer cancels an order while its export task runs. A worker cannot safely stop a handler at an
arbitrary point. Thus, a cancellation is a durable request. Each SDK gives the request to the
handler. The handler must check for the request and stop.

    ```ts
    worker.handle("export.build", async (payload: { parts: string[] }, context) => {
      for (const part of payload.parts) {
        if (context.signal.aborted) throw context.signal.reason;
        await uploadPart(part, { signal: context.signal });
      }
      return { exported: true };
    });
    ```

If no worker runs the task, Workhorse cancels it immediately. For example, the task waits in the
queue or in a durable wait. A task that never started gets no attempt history.

If a worker runs the task, the heartbeat of the worker gets the request. Then each SDK tells the
handler:

- TypeScript aborts `context.signal`.
- Python cancels `context.cancellation`.
- Go cancels the standard context of the handler.
- Rust reports the request through `context.cancellation()`.
- Ruby reports the request through `context.cancellation`.

If a handler ignores the request, the worker loses the task after some time. Then Workhorse records
the cancellation and does not retry the task.

## Build a durable agent loop

A model plans the work, and many tools run independently. The loop waits for an approval, and it
must continue after a deployment.

1. Put the model call in a checkpoint.
2. Run the tools as child tasks, and wait for all of them.
3. Wait on a durable timer.
4. Wait for an approval signal, which the application sends with an idempotency key.

<Tabs items={["TypeScript", "Python", "Go", "Rust", "Ruby"]}>
  <Tab value="TypeScript">
    ```ts
    worker.handle("agent.run", async ({ prompt, conversationId }, context) => {
      const plan = await context.checkpoint("plan", () =>
        model.plan(prompt, { idempotencyKey: `plan:${context.task.id}` }),
      );

      const tools = await context.runChildrenAll(
        plan.tools.map((tool) => ({
          name: tool.id,
          type: "agent.tool",
          payload: tool,
          options: { queue: "tools", concurrencyKey: conversationId },
        })),
      );

      await context.sleep("model-cooldown", cooldownMs);
      const approval = await context.waitForSignal<{ approved: boolean }>("approval");
      return { plan, tools, approved: approval.approved };
    });
    ```

    ```python verify
    from typing import Protocol, TypedDict, cast

    from workhorse import ChildTaskRequest, EnqueueOptions, HandlerContext, Json, Worker


    class AgentRun(TypedDict):
        prompt: str
        conversationId: str


    class Tool(TypedDict):
        id: str


    class Plan(TypedDict):
        tools: list[Tool]


    class Approval(TypedDict):
        approved: bool


    class Model(Protocol):
        def plan(self, prompt: str, *, idempotency_key: str) -> Json: ...


    def register_agent(worker: Worker, model: Model, cooldown_ms: int) -> None:
        def run_agent(payload: AgentRun, context: HandlerContext) -> dict[str, Json]:
            plan = context.checkpoint(
                "plan",
                lambda: model.plan(
                    payload["prompt"],
                    idempotency_key=f"plan:{context.task.id}",
                ),
            )

            tools = context.run_children_all(
                [
                    ChildTaskRequest(
                        name=tool["id"],
                        type="agent.tool",
                        payload=cast(Json, tool),
                        options=EnqueueOptions(
                            queue="tools",
                            concurrency_key=payload["conversationId"],
                        ),
                    )
                    for tool in cast(Plan, plan)["tools"]
                ]
            )

            context.sleep("model-cooldown", cooldown_ms)
            approval = cast(Approval, context.wait_for_signal("approval"))
            return {"plan": plan, "tools": tools, "approved": approval["approved"]}

        worker.handle("agent.run", run_agent)
    ```

    ```go verify
    package agent

    import (
    	"context"
    	"fmt"
    	"time"

    	workhorse "github.com/stablemates/workhorse/go"
    )

    type Model interface {
    	Plan(context.Context, string, string) (map[string]any, error)
    }

    func registerAgent(worker *workhorse.Worker, model Model, cooldown time.Duration) {
    	worker.Handle("agent.run", func(
    		ctx context.Context,
    		payload any,
    		handler *workhorse.HandlerContext,
    	) (any, error) {
    		message, ok := payload.(map[string]any)
    		if !ok {
    			return nil, fmt.Errorf("agent payload must be an object")
    		}
    		prompt, promptOK := message["prompt"].(string)
    		conversationID, conversationOK := message["conversationId"].(string)
    		if !promptOK || !conversationOK {
    			return nil, fmt.Errorf("agent payload needs string prompt and conversationId fields")
    		}

    		planValue, err := handler.Checkpoint("plan", func() (any, error) {
    			return model.Plan(ctx, prompt, "plan:"+handler.Task.ID)
    		})
    		if err != nil {
    			return nil, err
    		}
    		plan, ok := planValue.(map[string]any)
    		if !ok {
    			return nil, fmt.Errorf("model plan must be an object")
    		}
    		rawTools, ok := plan["tools"].([]any)
    		if !ok {
    			return nil, fmt.Errorf("model plan needs an array tools field")
    		}
    		children := make([]workhorse.ChildTaskRequest, 0, len(rawTools))
    		for _, value := range rawTools {
    			tool, ok := value.(map[string]any)
    			if !ok {
    				return nil, fmt.Errorf("tool must be an object")
    			}
    			toolID, ok := tool["id"].(string)
    			if !ok {
    				return nil, fmt.Errorf("tool needs a string id field")
    			}
    			children = append(children, workhorse.ChildTaskRequest{
    				Name:    toolID,
    				Type:    "agent.tool",
    				Payload: tool,
    				Options: workhorse.EnqueueOptions{
    					Queue:          "tools",
    					ConcurrencyKey: conversationID,
    				},
    			})
    		}

    		tools, err := handler.RunChildrenAll(children)
    		if err != nil {
    			return nil, err
    		}
    		if err := handler.Sleep("model-cooldown", cooldown); err != nil {
    			return nil, err
    		}
    		approval, err := handler.WaitForSignal("approval")
    		if err != nil {
    			return nil, err
    		}
    		approvalPayload, ok := approval.(map[string]any)
    		if !ok {
    			return nil, fmt.Errorf("approval must be an object")
    		}
    		approved, ok := approvalPayload["approved"].(bool)
    		if !ok {
    			return nil, fmt.Errorf("approval needs a boolean approved field")
    		}
    		return map[string]any{"plan": plan, "tools": tools, "approved": approved}, nil
    	})
    }
    ```

    ```rust
    use std::future::Future;
    use std::sync::Arc;
    use std::time::Duration;

    use serde::Deserialize;
    use serde_json::{json, Value};
    use workhorse::{ChildTaskRequest, EnqueueOptions, HandlerError, Worker};

    pub trait Model: Send + Sync + 'static {
        fn plan(
            &self,
            prompt: &str,
            idempotency_key: String,
        ) -> impl Future<Output = Result<Plan, HandlerError>> + Send;
    }

    #[derive(serde::Serialize, Deserialize)]
    pub struct Plan {
        tools: Vec<Tool>,
    }

    #[derive(serde::Serialize, Deserialize)]
    struct Tool {
        id: String,
        #[serde(flatten)]
        arguments: Value,
    }

    #[derive(Deserialize)]
    #[serde(rename_all = "camelCase")]
    struct AgentRun {
        prompt: String,
        conversation_id: String,
    }

    #[derive(Deserialize)]
    struct Approval {
        approved: bool,
    }

    pub fn register_agent<M: Model>(worker: &Worker, model: Arc<M>, cooldown: Duration) {
        worker.handle("agent.run", move |run: AgentRun, context| {
            let model = Arc::clone(&model);
            async move {
                let key = format!("plan:{}", context.task().id);
                let plan: Plan =
                    context.checkpoint("plan", || model.plan(&run.prompt, key)).await?;

                let mut children = Vec::with_capacity(plan.tools.len());
                for tool in &plan.tools {
                    let mut child = ChildTaskRequest::new(&tool.id, "agent.tool", tool)?;
                    child.options = EnqueueOptions {
                        queue: Some("tools".into()),
                        concurrency_key: Some(run.conversation_id.clone()),
                        ..Default::default()
                    };
                    children.push(child);
                }

                let tools = context.run_children_all(children).await?;
                context.sleep("model-cooldown", cooldown).await?;
                let approval: Approval = context.wait_for_signal("approval", None).await?.payload;
                Ok(json!({ "plan": plan, "tools": tools, "approved": approval.approved }))
            }
        });
    }
    ```
